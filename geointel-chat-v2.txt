// v2.0 changes vs v1.0:
//   - Two-phase reasoning (classify with Haiku, generate with Sonnet)
//   - Works on the full KG (entities + relations), not per-dossier brief_text
//   - Typed outputs: welcome, clarification, ready_to_generate, scenario, out_of_scope
//   - Tracks current scenario state across turns (memory)
//
// Required Supabase secret:
//   ANTHROPIC_API_KEY
//
// Request (POST JSON):
//   {
//     "question": string,              // user’s new message
//     "history": [                     // full chat history (user + assistant turns)
//       { "role": "user" | "assistant", "content": string }
//     ],
//     "kg": {                          // the full knowledge graph from data.js
//       "entities": [array of entity objects],
//       "relations": [array of relation objects]
//     },
//     "dossier_index": [               // light index of dossiers (id + title + description)
//       { "id": "taiwan-strait", "title": "Taiwan Strait", "description": "text" }
//     ],
//     "current_scenario": null | {     // what is currently rendered in Report + subgraph (null if nothing)
//       "question": string,
//       "entity_ids": [string],
//       "relation_keys": [string]      // keys are "from|to|type"
//     }
//   }
//
// Response (JSON, 200):
//   {
//     "type": "welcome" | "clarification" | "ready_to_generate" | "scenario" | "out_of_scope" | "acknowledge",
//     "message": string,               // text to append to the chat
//     "scenario"?: {                   // only when type === "scenario"
//       "title": string,
//       "report_html": string,
//       "entity_ids": [string],        // subset of KG to highlight
//       "relation_keys": [string]      // subset of KG to highlight (key = "from|to|type")
//     },
//     "debug"?: object
//   }

const ANTHROPIC_VERSION = "2023-06-01";
const MODEL_CLASSIFIER = "claude-haiku-4-5-20251001";
const MODEL_GENERATOR = "claude-sonnet-4-5-20250929";
const MAX_TOKENS_CLASSIFIER = 800;
const MAX_TOKENS_GENERATOR = 3000;
const MAX_HISTORY_TURNS = 30;

const CORS_HEADERS = {
"Access-Control-Allow-Origin": "*",
"Access-Control-Allow-Methods": "POST, OPTIONS",
"Access-Control-Allow-Headers": "Content-Type, Authorization",
"Access-Control-Max-Age": "86400",
};

// --------------------------------------------------
// HTTP HELPERS
// --------------------------------------------------
function jsonResponse(body: any, status = 200): Response {
const headers = Object.assign({ "Content-Type": "application/json" }, CORS_HEADERS);
return new Response(JSON.stringify(body), { status, headers });
}

function errorResponse(error: string, status: number, detail?: string): Response {
const body: Record<string, string> = { error };
if (detail) body.detail = detail;
return jsonResponse(body, status);
}

// --------------------------------------------------
// ANTHROPIC CALL
// --------------------------------------------------
async function callAnthropic(apiKey: string, model: string, maxTokens: number, systemPrompt: string, messages: any[]): Promise<{ ok: true; text: string; usage: any } | { ok: false; status: number; error: string }> {
let resp: Response;
try {
resp = await fetch(ANTHROPIC_API_URL, {
method: "POST",
headers: {
"Content-Type": "application/json",
"x-api-key": apiKey,
"anthropic-version": ANTHROPIC_VERSION,
},
body: JSON.stringify({
model,
max_tokens: maxTokens,
system: systemPrompt,
messages,
}),
});
} catch (e) {
return { ok: false, status: 502, error: "Network error: " + (e instanceof Error ? e.message : String(e)) };
}

const raw = await resp.text();
if (!resp.ok) {
return { ok: false, status: resp.status, error: "Anthropic API " + resp.status + ": " + raw.slice(0, 400) };
}

let parsed: any;
try {
parsed = JSON.parse(raw);
} catch {
return { ok: false, status: 502, error: "Anthropic returned non-JSON: " + raw.slice(0, 200) };
}

const blocks = Array.isArray(parsed.content) ? parsed.content : [];
const text = blocks
.filter((b: any) => b && b.type === "text" && typeof b.text === "string")
.map((b: any) => b.text)
.join("\n")
.trim();

if (!text) {
return { ok: false, status: 502, error: "Empty response from model" };
}
return { ok: true, text, usage: parsed.usage ?? null };
}

// --------------------------------------------------
// JSON PARSING WITH FALLBACKS
// --------------------------------------------------
function parseJsonLoose(text: string): any | null {
// Try direct parse
try {
return JSON.parse(text);
} catch {
// empty
}
// Try to strip markdown fences
const stripped = text.replace(/^`(?:json)?\s*/i, "").replace(/\s*`\s*$/i, "");
try {
return JSON.parse(stripped);
} catch {
// empty
}
// Try to find first { and last }
const first = text.indexOf("{");
const last = text.lastIndexOf("}");
if (first >= 0 && last > first) {
const slice = text.slice(first, last + 1);
try {
return JSON.parse(slice);
} catch {
// empty
}
}
return null;
}

// --------------------------------------------------
// INPUT BUILDERS
// --------------------------------------------------
function buildClassifierInput(req: any): any[] {
// Light entity index for the classifier (just id + label)
const entityIndex = (req.kg?.entities ?? []).map((e: any) => ({
id: e.id,
label: e.label,
type: e.type,
dossiers: e.dossiers,
}));

const history = (req.history ?? []).slice(-MAX_HISTORY_TURNS);
const historyText = history.map((t: any) => `[${t.role}] ${t.content}`).join("\n\n");

const payload = {
user_message: req.question,
chat_history: historyText || "(empty - first turn)",
dossiers_covered: req.dossier_index ?? [],
entity_index: entityIndex,
current_scenario: req.current_scenario ?? null,
};

return [
{
role: "user",
content: "Classify the following user interaction and respond with strict JSON as instructed.\n\n" + JSON.stringify(payload, null, 2),
},
];
}

function buildGeneratorInput(req: any, classifierResult: any): any[] {
const history = (req.history ?? []).slice(-MAX_HISTORY_TURNS);
const historyText = history.map((t: any) => `[${t.role}] ${t.content}`).join("\n\n");

const payload = {
user_question: req.question,
preselected_entity_ids: classifierResult.entity_ids ?? [],
kg: req.kg,
chat_history: historyText,
prior_scenario: req.current_scenario ?? null,
};

return [
{
role: "user",
content: "Generate a scenario projection report based on the following input. Respond with strict JSON as instructed.\n\n" + JSON.stringify(payload, null, 2),
},
];
}

// --------------------------------------------------
// MAIN HANDLER
// --------------------------------------------------
Deno.serve(async (req: Request) => {
if (req.method === "OPTIONS") {
return new Response(null, { status: 204, headers: CORS_HEADERS });
}
if (req.method !== "POST") {
return errorResponse("Method not allowed. Use POST.", 405);
}

const apiKey = Deno.env.get("ANTHROPIC_API_KEY");
if (!apiKey) {
return errorResponse("ANTHROPIC_API_KEY not configured", 500);
}

let payload: any;
try {
payload = await req.json();
} catch {
return errorResponse("Invalid JSON body", 400);
}

if (typeof payload.question !== "string") {
return errorResponse("‘question’ must be a string", 400);
}
if (!payload.kg || !Array.isArray(payload.kg.entities) || !Array.isArray(payload.kg.relations)) {
return errorResponse("‘kg’ must contain entities and relations arrays", 400);
}

// -- Phase A: classify --
const classifierMessages = buildClassifierInput(payload);
const classRes = await callAnthropic(apiKey, MODEL_CLASSIFIER, MAX_TOKENS_CLASSIFIER, CLASSIFIER_SYSTEM_PROMPT, classifierMessages);
if (!classRes.ok) {
return errorResponse("Classifier failure", classRes.status, classRes.error);
}

const classified = parseJsonLoose(classRes.text);
if (!classified || typeof classified.type !== "string" || typeof classified.message !== "string") {
return errorResponse("Classifier returned malformed JSON", 502, classRes.text.slice(0, 500));
}

// If the classifier type does not require generation, return now.
const generationTypes = new Set(["acknowledge"]);
if (!generationTypes.has(classified.type)) {
return jsonResponse({
type: classified.type,
message: classified.message,
debug: { classifier: classified.reasoning_note ?? null, classifier_usage: classRes.usage },
});
}

// -- Phase B: generate scenario --
// For "acknowledge", we need to figure out what scenario to generate.
// The scenario question and entity_ids come from the PREVIOUS assistant turn (ready_to_generate / scenario_followup).
// The classifier should have passed entity_ids in its output, OR we need to extract from history.
// Safer: ask the classifier to include entity_ids also for "acknowledge" by looking at prior turns.
// If missing, fall back to re-classifying the implied question.

let entity_ids = Array.isArray(classified.entity_ids) ? classified.entity_ids : [];
let impliedQuestion = payload.question;

// If entity_ids empty on acknowledge, try to extract from history: the last user message before the ack
if (entity_ids.length === 0 && Array.isArray(payload.history)) {
for (let i = payload.history.length - 1; i >= 0; i–) {
const t = payload.history[i];
if (t.role === "user" && t.content && t.content.trim() !== payload.question.trim()) {
impliedQuestion = t.content;
break;
}
}
}

const genInput = buildGeneratorInput(
Object.assign({}, payload, { question: impliedQuestion }),
{ entity_ids }
);

const genRes = await callAnthropic(apiKey, MODEL_GENERATOR, MAX_TOKENS_GENERATOR, GENERATOR_SYSTEM_PROMPT, genInput);
if (!genRes.ok) {
return errorResponse("Generator failure", genRes.status, genRes.error);
}

const scenario = parseJsonLoose(genRes.text);
if (!scenario || typeof scenario.report_html !== "string" || !Array.isArray(scenario.entity_ids) || !Array.isArray(scenario.relation_keys)) {
return errorResponse("Generator returned malformed JSON", 502, genRes.text.slice(0, 500));
}

return jsonResponse({
type: "scenario",
message: classified.message, // short confirmation in chat: "Procedo con la generazione." or similar
scenario: {
title: scenario.title || "Scenario Projection",
report_html: scenario.report_html,
entity_ids: scenario.entity_ids,
relation_keys: scenario.relation_keys,
},
debug: {
classifier_usage: classRes.usage,
generator_usage: genRes.usage,
classifier_note: classified.reasoning_note ?? null,
},
});
});
